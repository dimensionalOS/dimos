cmake_minimum_required(VERSION 3.20)
project(external_native_relay LANGUAGES CXX)
find_package(external_telemetry CONFIG REQUIRED)
if(NOT DEFINED DIMOS_NATIVE_CPP_DIR)
  set(DIMOS_NATIVE_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../native/cpp")
endif()
add_subdirectory(${DIMOS_NATIVE_CPP_DIR} ${CMAKE_BINARY_DIR}/dimos_native)
add_executable(external_native_relay relay.cpp)
target_link_libraries(external_native_relay PRIVATE dimos_native external_telemetry::messages)
