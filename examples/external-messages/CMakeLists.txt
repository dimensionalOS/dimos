cmake_minimum_required(VERSION 3.20)
project(acme_telemetry_consumer LANGUAGES CXX)
find_package(DimosMessages CONFIG REQUIRED)
dimos_message_project()
add_executable(telemetry_relay cpp/relay.cpp)
target_link_libraries(telemetry_relay PRIVATE acme_telemetry::messages)

option(DIMOS_BUILD_MODULE_EXAMPLE "Build the real CDR SDK module" OFF)
if(DIMOS_BUILD_MODULE_EXAMPLE)
  dimos_native_sdk()
  add_executable(telemetry_module modules/cpp.cpp)
  target_link_libraries(telemetry_module PRIVATE dimos_native acme_telemetry::messages)
endif()
